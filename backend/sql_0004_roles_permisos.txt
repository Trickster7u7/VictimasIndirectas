BEGIN;
--
-- Create model Rol
--
CREATE TABLE "roles" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "codigo" VARCHAR(50) NOT NULL UNIQUE,
    "nombre" VARCHAR(100) NOT NULL,
    "descripcion" TEXT,
    "activo" BOOL NOT NULL,
    "es_sistema" BOOL NOT NULL,
    "fecha_creacion" TIMESTAMPTZ NOT NULL,
    "fecha_actualizacion" TIMESTAMPTZ NOT NULL
);
--
-- Create model Permiso
--
CREATE TABLE "permisos" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "codigo" VARCHAR(100) NOT NULL UNIQUE,
    "nombre" VARCHAR(120) NOT NULL,
    "modulo" VARCHAR(50) NOT NULL,
    "descripcion" TEXT
);
--
-- Create model UsuarioRol
--
CREATE TABLE "usuario_roles" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "fecha_asignacion" TIMESTAMPTZ NOT NULL,
    "asignado_por_id" INT REFERENCES "usuarios" ("id") ON DELETE SET NULL,
    "rol_id" INT NOT NULL REFERENCES "roles" ("id") ON DELETE RESTRICT,
    "usuario_id" INT NOT NULL REFERENCES "usuarios" ("id") ON DELETE CASCADE,
    CONSTRAINT "uid_usuario_rol_usuario_2d79b9" UNIQUE ("usuario_id", "rol_id")
);
--
-- Create model RolPermiso
--
CREATE TABLE "rol_permisos" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "permiso_id" INT NOT NULL REFERENCES "permisos" ("id") ON DELETE RESTRICT,
    "rol_id" INT NOT NULL REFERENCES "roles" ("id") ON DELETE CASCADE,
    CONSTRAINT "uid_rol_permiso_rol_id_6f43c4" UNIQUE ("rol_id", "permiso_id")
);
--
-- Create model AuditoriaRoles
--
CREATE TABLE "auditoria_roles" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "roles_anteriores" JSONB NOT NULL,
    "roles_nuevos" JSONB NOT NULL,
    "fecha_cambio" TIMESTAMPTZ NOT NULL,
    "actor_id" INT NOT NULL REFERENCES "usuarios" ("id") ON DELETE RESTRICT,
    "usuario_id" INT NOT NULL REFERENCES "usuarios" ("id") ON DELETE RESTRICT
);
--
-- Create model Grupo
--
CREATE TABLE "grupos" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "codigo" VARCHAR(50) NOT NULL UNIQUE,
    "nombre" VARCHAR(120) NOT NULL,
    "descripcion" TEXT,
    "activo" BOOL NOT NULL,
    "fecha_creacion" TIMESTAMPTZ NOT NULL,
    "fecha_actualizacion" TIMESTAMPTZ NOT NULL
);
--
-- Create model UsuarioGrupo
--
CREATE TABLE "usuario_grupos" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "fecha_asignacion" TIMESTAMPTZ NOT NULL,
    "asignado_por_id" INT REFERENCES "usuarios" ("id") ON DELETE SET NULL,
    "grupo_id" INT NOT NULL REFERENCES "grupos" ("id") ON DELETE RESTRICT,
    "usuario_id" INT NOT NULL REFERENCES "usuarios" ("id") ON DELETE CASCADE,
    CONSTRAINT "uid_usuario_gru_usuario_d6502a" UNIQUE ("usuario_id", "grupo_id")
);
--
-- Create model GrupoRol
--
CREATE TABLE "grupo_roles" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "grupo_id" INT NOT NULL REFERENCES "grupos" ("id") ON DELETE CASCADE,
    "rol_id" INT NOT NULL REFERENCES "roles" ("id") ON DELETE RESTRICT,
    CONSTRAINT "uid_grupo_roles_grupo_i_178040" UNIQUE ("grupo_id", "rol_id")
);
--
-- Create model AuditoriaGrupos
--
CREATE TABLE "auditoria_grupos" (
    "id" SERIAL NOT NULL PRIMARY KEY,
    "tipo_cambio" VARCHAR(30) NOT NULL,
    "ids_anteriores" JSONB NOT NULL,
    "ids_nuevos" JSONB NOT NULL,
    "fecha_cambio" TIMESTAMPTZ NOT NULL,
    "actor_id" INT NOT NULL REFERENCES "usuarios" ("id") ON DELETE RESTRICT,
    "grupo_id" INT REFERENCES "grupos" ("id") ON DELETE RESTRICT,
    "usuario_id" INT REFERENCES "usuarios" ("id") ON DELETE RESTRICT
);
COMMIT;
